# INSTALL ON THE PI ONLY:
#     python -m pip install -e . -r requirements-pi.txt
#
# Every pin below has a Python 3.13 manylinux aarch64 wheel on PyPI (checked
# 2026-09-29), and the stack was rehearsed under Python 3.13 on the dev PC: all
# unit tests pass, the Tasks backend loads the .task bundles, Vosk loads, and the
# INT8 model gives IDENTICAL outputs under ai-edge-litert 2.2.0 and tf.lite on all
# 849 recorded windows.
#
# picamera2 is NOT installed from pip: python3-picamera2 + python3-libcamera come
# from APT (built for the system Python), so create the venv with
#     python3 -m venv --system-site-packages .venv
# ---------------------------------------------------------------------------

numpy>=1.26               # the landmark math is version-agnostic; let the resolver pick
vosk==0.3.45              # small en-us model is downloaded separately into models/

# TRANSFER-CRITICAL -- MediaPipe. Training used 0.10.35 on the PC; 1.0.x is the only
# line with Python 3.13 aarch64 wheels. It keeps the Tasks API we use
# (HandLandmarker/PoseLandmarker); legacy Holistic is gone (the code reports that
# clearly). Landmark values come from the .task model bundles, so copy models/*.task
# over from the dev box -- don't re-download a different variant.
# mediapipe pulls in opencv-contrib-python (cv2); do NOT also install opencv-python,
# the two packages overwrite each other's cv2 module.
mediapipe==1.0.1
sounddevice~=0.5          # mediapipe 1.0 requires ~=0.5; needs: sudo apt install libportaudio2

# TFLite inference runtime (LiteRT). 2.x is the first line with cp313 wheels; it
# still provides ai_edge_litert.interpreter.Interpreter, which SignClassifier uses.
ai-edge-litert==2.2.0
